// ---------- helpers ----------
const q = <T extends Element = HTMLElement>(root: ParentNode, sel: string) => { const e = root.querySelector<T>(sel); if (!e) throw new Error(`missing ${sel}`); return e; };
const qa = (root: ParentNode, sel: string) => [...root.querySelectorAll<HTMLElement>(sel)];
const byText = (root: ParentNode, sel: string, text: string | RegExp) => {
  const found = qa(root, sel).find(e => typeof text === "string" ? e.textContent?.trim() === text : text.test(e.textContent ?? ""));
  if (!found) throw new Error(`missing ${sel} "${text}"`);
  return found;
};
const html = (node: JSX.Element) => renderToStaticMarkup(node);
const click = (el: Element, n: number | string) => { el.classList.add("ux-click"); el.setAttribute("data-click", String(n)); };
const isNew = (el: Element) => { el.classList.add("ux-new"); return el; };
async function snap(node: JSX.Element, ready: () => Promise<unknown>, selector?: string, after?: () => Promise<void> | void): Promise<HTMLElement> {
  render(node);
  await ready();
  if (after) await after();
  qa(document.body, "input").forEach(i => { const e = i as HTMLInputElement; if (e.checked) e.setAttribute("checked", ""); else e.removeAttribute("checked"); if (e.type !== "checkbox" && e.type !== "radio") e.setAttribute("value", e.value); });
  qa(document.body, "select").forEach(sel => [...(sel as HTMLSelectElement).options].forEach(o => o.selected ? o.setAttribute("selected", "") : o.removeAttribute("selected")));
  qa(document.body, "textarea").forEach(t => { t.textContent = (t as HTMLTextAreaElement).value; });
  const el = (selector ? q(document.body, selector) : document.body).cloneNode(true) as HTMLElement;
  cleanup();
  return el;
}
const dialogFrame = (el: HTMLElement | string) => `<div class="ux-dialog">${typeof el === "string" ? el : el.outerHTML}</div>`;
const pad = (inner: string, cls = "") => `<div class="ux-pad ${cls}">${inner}</div>`;
const state = (label: string, body: string) => `<p class="dn-state">${label}</p>${body}`;
const shot = (path: string, alt: string, cls = "") => `<img class="ux-shot ${cls}" src="${path}" alt="${alt}" loading="lazy">`;
/** Only the details pane of a CaseScreen render, at its real width, without the fixed height. */
const paneOf = (body: HTMLElement) => `<div class="ux-pane">${q(body, ".inspector-pane").outerHTML}</div>`;
const appFrame = (body: HTMLElement, cls = "") => `<div class="ux-app ${cls}">${body.firstElementChild!.outerHTML}</div>`;

interface Option { letter: string; name: string; recommended?: string; clicks?: string; note?: string; html: string }
interface Section { id: string; title: string; question: string; finding: string; today: { html: string; clicks?: string; caption: string }; options: Option[]; wide?: boolean; extra?: string }

function option(o: Option): string {
  return `<article class="dn-option${o.recommended ? " dn-option--rec" : ""}" aria-label="Option ${o.letter}">
    <h3 class="dn-option-head"><span class="dn-letter">${o.letter}</span> ${o.name}${o.recommended ? ` <span class="dn-rec">recommended</span>` : ""}</h3>
    ${o.recommended ? `<p class="ux-why"><strong>Why:</strong> ${o.recommended}</p>` : ""}
    ${o.clicks ? `<p class="ux-clicks">${o.clicks}</p>` : ""}
    ${o.note ? `<p class="dn-note">${o.note}</p>` : ""}
    <div class="dn-frame">${o.html}</div>
  </article>`;
}
function section(s: Section): string {
  const rec = s.options.find(o => o.recommended)!;
  return `<section class="dn dn--open" id="${s.id}" data-status="open" aria-labelledby="${s.id}-h">
  <h2 id="${s.id}-h" class="dn-title">${s.title} <span class="dn-status dn-status--open">open, recommended ${rec.letter}</span></h2>
  <p class="dn-finding">${s.finding}</p>
  <p class="dn-question">${s.question}</p>
  <div class="ux-today"><h3 class="ux-today-h">Today${s.today.clicks ? ` <span class="ux-today-clicks">${s.today.clicks}</span>` : ""}</h3><p class="dn-note">${s.today.caption}</p>${s.today.html}</div>
  <div class="dn-options${s.wide ? " dn-options--stacked" : ""}">
  ${s.options.map(option).join("\n")}
  </div>
  ${s.extra ?? ""}
</section>`;
}

interface TourStep { img: string; cap: string; dns?: string[] }
const TOUR: { id: string; title: string; who: string; steps: TourStep[] }[] = [];

function writePages(): void {
  const css = "app/assets/" + readdirSync(resolve(OUT, "app/assets")).find(f => f.endsWith(".css"));
  const head = (title: string) => `<!doctype html>
<html lang="en">
<head>
<meta charset="utf-8">
<meta name="viewport" content="width=device-width, initial-scale=1">
<title>${title}</title>
<link rel="stylesheet" href="${css}">
<link rel="stylesheet" href="mockups.css">
</head>`;
  const plain = (t: string) => t.replace(/<[^>]+>/g, "");
  const openList = `<section class="dn-open-list" aria-labelledby="open-h"><h2 id="open-h">Open questions</h2><ul>${SECTIONS.map(s => {
    const rec = s.options.find(o => o.recommended)!;
    return `<li><a href="#${s.id}">${s.title.split(" ")[0]}</a> ${s.title.replace(/^\S+ /, "")} <span class="dn-status dn-status--open">open, recommended ${rec.letter}: ${plain(rec.name)}</span></li>`;
  }).join("")}</ul><p>Nothing is decided yet. Each section shows today, then the options side by side with the recommended one outlined in blue.</p></section>`;
  writeFileSync(resolve(OUT, "index.html"), `${head("M2 acceptance: decisions M-1 … M-8 (local mockups)")}
<body>
<main class="dn-page">
<header class="dn-intro">
  <p><a href="tour.html">Tour: where each decision appears, lead → annotator → consolidation → settings →</a></p>
  <h1>Manual acceptance M2: UI decisions M-1 … M-8</h1>
  ${openList}
  <p>Local mockups for the open UI questions of <code>build/m2/FINDINGS-2026-10-26.md</code> and <code>build/m2/settings-audit/REPORT.md</code> (feat/ux-calm). Option frames are the real components rendered in jsdom with the built app CSS; only what an option changes is edited and marked with a dashed outline. Red outlines with a number are counted clicks. “Today” frames are rendered from the same components or are acceptance screenshots.</p>
  <nav aria-label="Decisions"><ul class="dn-nav">${SECTIONS.map(s => `<li><a href="#${s.id}">${s.title.split(" ")[0]}</a></li>`).join("")}</ul></nav>
</header>
${SECTIONS.map(section).join("\n")}
</main>
</body>
</html>`);
  writeTour(head);
}

function writeTour(head: (t: string) => string): void {
  if (!TOUR.length) return;
  const ids = new Set(SECTIONS.map(s => s.id));
  const titleOf = (id: string) => SECTIONS.find(s => s.id === id)?.title.replace(/^\S+ /, "") ?? id;
  const badge = (id: string) => `<a class="tour-dn" href="index.html#${id.split("/")[0]}" title="${titleOf(id.split("/")[0]).replace(/"/g, "&quot;")}">open: ${id.toUpperCase()}</a>`;
  const label = (flow: number, i: number) => `${flow + 1}.${i + 1}`;
  const where = [...ids].map(id => {
    const steps = TOUR.flatMap((f, fi) => f.steps.map((st, i) => ({ f, fi, i, st }))).filter(x => x.st.dns?.some(d => d.split("/")[0] === id));
    return `<li><a href="index.html#${id}">${id.toUpperCase()}</a> ${titleOf(id)}: step ${steps.map(x => `<a href="#${x.f.id}-${x.i + 1}">${label(x.fi, x.i)}</a>`).join(", ") || "none"}</li>`;
  }).join("");
  writeFileSync(resolve(OUT, "tour.html"), `${head("M2 acceptance: tour (local mockups)")}
<body><main class="dn-page tour-page">
<header class="dn-intro"><p><a href="index.html">← Decisions M-1 … M-8 (options side by side)</a></p>
<h1>M2 acceptance: where each decision shows up</h1>
<p>The acceptance in the order it was walked: lead on Documents, annotator in the editor, lead consolidating, lead in Settings. Images are acceptance screenshots (build/m2) or the “today” frames rendered for index.html; click to enlarge. Orange steps are where an open decision appears; the badge links to its options.</p>
<section class="dn-open-list" aria-labelledby="open-h"><h2 id="open-h">Open questions and their steps</h2><ul>${where}</ul></section>
</header>
${TOUR.map((f, fi) => `<section class="tour-flow" id="${f.id}" aria-labelledby="${f.id}-h"><h2 id="${f.id}-h">${fi + 1}. ${f.title}</h2><p class="dn-note">${f.who}</p>
<ol class="tour-steps">${f.steps.map((st, i) => `<li class="tour-step${st.dns ? " tour-step--decision" : ""}" id="${f.id}-${i + 1}"><span class="tour-num">${i + 1}</span>
<p class="tour-cap">${st.cap}</p><a href="${st.img}"><img src="${st.img}" alt="Step ${label(fi, i)}" loading="lazy"></a>
${st.dns ? `<p class="tour-dns">${st.dns.map(badge).join("")}</p>` : ""}</li>`).join("\n")}</ol></section>`).join("\n")}
</main></body></html>`);
}
