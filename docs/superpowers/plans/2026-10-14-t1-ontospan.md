> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A lightweight Python package `ontospan` that defines, validates, reads and writes the shared span annotation format, compares annotation sets, and exports Phenopackets.

**Architecture:**
- Pydantic models describe the format. JSON Schemas are generated from them and committed; they are the binding contract (T1 F12).
- Validation runs in two passes: the strict JSON Schema, then content checks (canonical text, hashes, offsets) and profile rules (`generic`, `hpo-phenotype`).
- Ontology knowledge enters only through a small `OntologyView` protocol. A stdlib loader for OBO Graph JSON is included.
- Matching, bundles and the Phenopacket export build on the models.

Spec: `docs/superpowers/specs/2026-10-14-t1-shared-annotation-format-design.md` in the Ontocurator repository.

**Tech Stack:** Python ≥ 3.11, Pydantic 2, jsonschema 4, pytest, ruff, mypy (strict), uv, hatchling. Optional extra: `phenopackets` + `protobuf`.

```markdown
# ontospan

Shared format for ontology concept annotations on text spans, used by
Ontocurator, Phentrieve and phentrieve-benchmark.

- Canonical text (NFC, LF) with Unicode codepoint offsets.
- One annotated document = one text + one annotation set; bundles group them.
- Profiles: `generic` and `hpo-phenotype` (axes, modifiers, allowed roots).
- Matching and agreement rules shared by every tool.
- Phenopacket v2 export (`pip install ontospan[phenopacket]` adds validation).

The JSON Schemas in `src/ontospan/schema/` are the binding contract.
Regenerate them with `uv run python -m ontospan.schema_export`.
```

```python
"""The shared span annotation format (schema version 1.0.0).

Models ignore unknown fields so that readers accept newer minor versions
(T1 F3). Strictness for the declared version comes from the generated JSON
Schema, which closes every object (see ``schema_export``).
"""

from __future__ import annotations

from datetime import datetime
from typing import Literal

from pydantic import BaseModel, ConfigDict, Field, model_validator

CURIE = r"^[A-Za-z][A-Za-z0-9_.]*:\S+$"
SHA256 = r"^[0-9a-f]{64}$"
LANGUAGE = r"^[a-z]{2,3}(-[A-Za-z0-9]+)*$"
ORCID = r"^\d{4}-\d{4}-\d{4}-\d{3}[\dX]$"

Basis = Literal["chosen", "defaulted"]
ActorKind = Literal["human", "tool"]
Method = Literal["manual", "adopted", "edited", "merged", "tool"]
SetKind = Literal["individual", "stage_result", "adjudicated", "release", "tool_output"]


class _Model(BaseModel):
    model_config = ConfigDict(extra="ignore", frozen=True)


class Segment(_Model):
    """A codepoint range [start, end) of the canonical text and the text it quotes."""

    start: int = Field(ge=0)
    end: int = Field(gt=0)
    text: str = Field(min_length=1)

    @model_validator(mode="after")
    def _consistent(self) -> Segment:
        if self.end <= self.start:
            raise ValueError("segment end must be greater than start")
        if len(self.text) != self.end - self.start:
            raise ValueError("segment text length must equal end - start")
        return self


class Mention(_Model):
    """One place a concept is mentioned; several segments make it discontinuous."""

    segments: tuple[Segment, ...] = Field(min_length=1)
    cue_segments: tuple[Segment, ...] = ()


class AxisValue(_Model):
    value: str = Field(min_length=1)
    basis: Basis


class Modifier(_Model):
    term_id: str = Field(pattern=CURIE)
    segments: tuple[Segment, ...] = ()


class Actor(_Model):
    actor_id: str = Field(min_length=1)
    kind: ActorKind
    name: str | None = None
    version: str | None = None
    model: str | None = None
    run_id: str | None = None


class Provenance(_Model):
    actor: Actor
    method: Method
    source_refs: tuple[str, ...] = ()
    notes: tuple[str, ...] = ()


class Annotation(_Model):
    annotation_id: str = Field(min_length=1)
    layer_id: str = Field(min_length=1)
    term_id: str = Field(pattern=CURIE)
    axes: dict[str, AxisValue] = Field(default_factory=dict)
    mentions: tuple[Mention, ...] = ()
    document_level: bool = False
    modifiers: tuple[Modifier, ...] = ()
    provenance: Provenance


class OntologyResource(_Model):
    id: str = Field(min_length=1)
    version: str = Field(min_length=1)
    sha256: str | None = Field(default=None, pattern=SHA256)
    iri: str | None = None


class Layer(_Model):
    layer_id: str = Field(min_length=1)
    profile: str = Field(min_length=1)
    ontology: OntologyResource
    allowed_roots: tuple[str, ...] = ()


class AnnotationSet(_Model):
    annotation_set_id: str = Field(min_length=1)
    set_kind: SetKind
    stage: str | None = None
    actors: tuple[Actor, ...] = Field(min_length=1)
    created_at: datetime
    blind: bool | None = None
    guideline_ref: str | None = None
    layers: tuple[Layer, ...] = Field(min_length=1)
    annotations: tuple[Annotation, ...] = ()


class Document(_Model):
    document_id: str = Field(min_length=1)
    language: str = Field(pattern=LANGUAGE)
    text: str
    text_sha256: str = Field(pattern=SHA256)
    source_text_sha256: str | None = Field(default=None, pattern=SHA256)
    subject_id: str | None = None


class AnnotatedDocument(_Model):
    """The atomic unit: one canonical text and exactly one annotation set."""

    schema_version: str = Field(pattern=r"^\d+\.\d+\.\d+$")
    artifact_type: Literal["annotated_document"]
    document: Document
    annotation_set: AnnotationSet


class Contributor(_Model):
    actor_id: str = Field(min_length=1)
    name: str | None = None
    orcid: str | None = Field(default=None, pattern=ORCID)


class FileEntry(_Model):
    path: str = Field(min_length=1)
    sha256: str = Field(pattern=SHA256)


class Manifest(_Model):
    schema_version: str = Field(pattern=r"^\d+\.\d+\.\d+$")
    artifact_type: Literal["bundle"]
    bundle_id: str = Field(min_length=1)
    release_name: str | None = None
    created_at: datetime
    layers: tuple[Layer, ...] = Field(min_length=1)
    contributors: tuple[Contributor, ...] = ()
    files: tuple[FileEntry, ...] = Field(min_length=1)
```

```python
"""Canonical text, hashes and the construction of documents and segments (T1 F2)."""

from __future__ import annotations

import hashlib
import unicodedata

from ontospan.models import Document, Segment


def canonical_text(raw: str) -> str:
    """NFC with LF line endings: the only text that offsets refer to."""
    return unicodedata.normalize("NFC", raw.replace("\r\n", "\n").replace("\r", "\n"))


def text_sha256(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def make_document(
    document_id: str,
    language: str,
    raw: str | bytes,
    *,
    subject_id: str | None = None,
) -> Document:
    """A document with canonical text; ``source_text_sha256`` keeps the original.

    Bytes must be UTF-8; a BOM is dropped. Other encodings raise
    ``UnicodeDecodeError`` instead of being guessed.
    """
    if isinstance(raw, bytes):
        source_sha = hashlib.sha256(raw).hexdigest()
        decoded = raw.decode("utf-8-sig")
    else:
        source_sha = text_sha256(raw)
        decoded = raw
    text = canonical_text(decoded)
    return Document(
        document_id=document_id,
        language=language,
        text=text,
        text_sha256=text_sha256(text),
        source_text_sha256=source_sha,
        subject_id=subject_id,
    )


def segment(text: str, start: int, end: int) -> Segment:
    """The segment [start, end) of a canonical text, quoting it."""
    return Segment(start=start, end=end, text=text[start:end])
```

```python
"""What the format needs to know about an ontology, and a loader for OBO Graph JSON.

Tools pass their own ontology through ``OntologyView``; ``ObographOntology``
is a small stdlib implementation for files such as ``hp.json``.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Protocol


class OntologyView(Protocol):
    def label(self, term_id: str) -> str | None: ...

    def is_active(self, term_id: str) -> bool: ...

    def ancestors(self, term_id: str) -> frozenset[str]:
        """Every term above ``term_id`` (``is_a``, transitive), without itself."""
        ...


def is_under(ontology: OntologyView, term_id: str, root: str) -> bool:
    """True if the term is the root or one of its descendants."""
    return term_id == root or root in ontology.ancestors(term_id)


def _curie(iri: str) -> str:
    local = iri.rsplit("/", 1)[-1]
    prefix, sep, rest = local.partition("_")
    return f"{prefix}:{rest}" if sep else local


class ObographOntology:
    def __init__(
        self,
        labels: dict[str, str],
        parents: dict[str, frozenset[str]],
        obsolete: frozenset[str],
        version: str | None,
    ) -> None:
        self._labels = labels
        self._parents = parents
        self._obsolete = obsolete
        self.version = version
        self._ancestors: dict[str, frozenset[str]] = {}

    @classmethod
    def load(cls, path: Path | str) -> ObographOntology:
        graph = json.loads(Path(path).read_text(encoding="utf-8"))["graphs"][0]
        labels: dict[str, str] = {}
        obsolete: set[str] = set()
        parents: dict[str, set[str]] = {}
        for node in graph.get("nodes", []):
            if node.get("type", "CLASS") != "CLASS":
                continue
            term = _curie(node["id"])
            labels[term] = node.get("lbl") or term
            if node.get("meta", {}).get("deprecated"):
                obsolete.add(term)
        for edge in graph.get("edges", []):
            if edge.get("pred") == "is_a":
                parents.setdefault(_curie(edge["sub"]), set()).add(_curie(edge["obj"]))
        return cls(
            labels,
            {term: frozenset(above) for term, above in parents.items()},
            frozenset(obsolete),
            graph.get("meta", {}).get("version"),
        )

    def label(self, term_id: str) -> str | None:
        return self._labels.get(term_id)

    def is_active(self, term_id: str) -> bool:
        return term_id in self._labels and term_id not in self._obsolete

    def ancestors(self, term_id: str) -> frozenset[str]:
        known = self._ancestors.get(term_id)
        if known is not None:
            return known
        seen: set[str] = set()
        stack = list(self._parents.get(term_id, ()))
        while stack:
            parent = stack.pop()
            if parent not in seen:
                seen.add(parent)
                stack.extend(self._parents.get(parent, ()))
        seen.discard(term_id)
        result = frozenset(seen)
        self._ancestors[term_id] = result
        return result
```

```python
"""Segment normalisation (T1 F11) and helpers for discontinuous mentions (T1 F4)."""

from __future__ import annotations

import unicodedata
from collections.abc import Iterable

from ontospan.models import Mention, Segment


def _trimmable(char: str) -> bool:
    return char.isspace() or unicodedata.category(char).startswith("P")


def normalize_segment(text: str, seg: Segment) -> tuple[int, int] | None:
    """The segment without edge whitespace and punctuation; None if nothing is left."""
    start, end = seg.start, seg.end
    while start < end and _trimmable(text[start]):
        start += 1
    while end > start and _trimmable(text[end - 1]):
        end -= 1
    return (start, end) if start < end else None


def spanning_segment(mention: Mention, text: str) -> Segment:
    """One segment from the first to the last segment of a mention."""
    start = min(seg.start for seg in mention.segments)
    end = max(seg.end for seg in mention.segments)
    return Segment(start=start, end=end, text=text[start:end])


def segments_flat(mentions: Iterable[Mention]) -> list[Segment]:
    """Every segment of these mentions, in text order."""
    return sorted(
        (seg for mention in mentions for seg in mention.segments),
        key=lambda seg: (seg.start, seg.end),
    )
```

```python
"""Generate the JSON Schemas from the models; ``--check`` fails on drift.

Run ``python -m ontospan.schema_export`` after changing a model.
"""

from __future__ import annotations

import json
import sys
from pathlib import Path
from typing import Any

from ontospan.models import AnnotatedDocument, Manifest

SCHEMA_DIR = Path(__file__).parent / "schema"
DIALECT = "https://json-schema.org/draft/2020-12/schema"
SCHEMA_FILES = ("annotated-document-1.0.json", "bundle-manifest-1.0.json")


def _close(node: Any) -> Any:
    """Forbid unknown properties on every object that lists its properties."""
    if isinstance(node, dict):
        if "properties" in node and "additionalProperties" not in node:
            node["additionalProperties"] = False
        for value in node.values():
            _close(value)
    elif isinstance(node, list):
        for value in node:
            _close(value)
    return node


def _schema(model: type[AnnotatedDocument] | type[Manifest], name: str) -> dict[str, Any]:
    schema: dict[str, Any] = _close(model.model_json_schema())
    return {"$schema": DIALECT, "$id": f"https://ontospan.local/schema/{name}", **schema}


def build_schemas() -> dict[str, dict[str, Any]]:
    return {
        SCHEMA_FILES[0]: _schema(AnnotatedDocument, SCHEMA_FILES[0]),
        SCHEMA_FILES[1]: _schema(Manifest, SCHEMA_FILES[1]),
    }


def _render(schema: dict[str, Any]) -> str:
    return json.dumps(schema, indent=2, ensure_ascii=False, sort_keys=True) + "\n"


def main(argv: list[str] | None = None) -> int:
    check = "--check" in (argv if argv is not None else sys.argv[1:])
    stale = []
    for name, schema in build_schemas().items():
        target = SCHEMA_DIR / name
        rendered = _render(schema)
        if check:
            if not target.is_file() or target.read_text(encoding="utf-8") != rendered:
                stale.append(name)
        else:
            target.write_text(rendered, encoding="utf-8", newline="\n")
    if stale:
        print(f"Stale schemas: {', '.join(stale)}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
import json
from collections.abc import Iterable, Mapping, Sequence
from functools import cache
from importlib import resources

import jsonschema
from pydantic import BaseModel, ValidationError

def schema_validator(name: str) -> jsonschema.Draft202012Validator:
    text = resources.files("ontospan.schema").joinpath(name).read_text(encoding="utf-8")
    return jsonschema.Draft202012Validator(json.loads(text))


def _path(parts: Iterable[Any]) -> str:
    return ".".join(str(part) for part in parts) or "$"


def check_version(version: object, report: ValidationReport) -> bool:
    """Record version problems; False if the data must not be read at all."""
    parts = version.split(".") if isinstance(version, str) else []
    if len(parts) != 3 or not all(part.isdigit() for part in parts):
        report.add("error", "schema_version", "schema_version", f"invalid version {version!r}")
        return False
    major, minor = int(parts[0]), int(parts[1])
    if major != SUPPORTED_MAJOR:
        report.add("error", "unsupported_major", "schema_version", f"major version {major} is not supported")
        return False
    if minor > SUPPORTED_MINOR:
        report.add(
            "warning",
            "newer_minor",
            "schema_version",
            f"version {version} is newer than {SUPPORTED_MAJOR}.{SUPPORTED_MINOR}; unknown fields are ignored",
        )
    return True


def check_schema(data: Mapping[str, Any], name: str, report: ValidationReport) -> None:
    """Strict JSON Schema check, skipped for newer minor versions."""
    if any(issue.code == "newer_minor" for issue in report.issues):
        return
    for error in sorted(schema_validator(name).iter_errors(data), key=lambda e: list(e.absolute_path)):
        report.add("error", "schema", _path(error.absolute_path), error.message)


def _parse(data: Mapping[str, Any], report: ValidationReport) -> AnnotatedDocument | None:
    if not check_version(data.get("schema_version"), report):
        return None
    check_schema(data, DOCUMENT_SCHEMA, report)
    if not report.ok:
        return None
    try:
        return AnnotatedDocument.model_validate(data)
    except ValidationError as exc:
        for error in exc.errors():
            report.add("error", "model", _path(error["loc"]), error["msg"])
        return None


def _check_segments(segments: Sequence[Segment], text: str, path: str, report: ValidationReport) -> None:
    previous_end = -1
    for index, seg in enumerate(segments):
        where = f"{path}[{index}]"
        if seg.end > len(text):
            report.add("error", "segment_out_of_range", where, f"segment ends at {seg.end}, text has {len(text)} codepoints")
            continue
        quoted = text[seg.start : seg.end]
        if quoted != seg.text:
            report.add("error", "segment_text_mismatch", where, f"expected {quoted!r}, got {seg.text!r}")
        if seg.start < previous_end:
            report.add("error", "segments_unordered", where, "segments must be sorted and must not overlap")
        previous_end = seg.end


def _check_content(
    unit: AnnotatedDocument, ontologies: Mapping[str, OntologyView], report: ValidationReport
) -> None:
    document = unit.document
    text = document.text
    if canonical_text(text) != text:
        report.add("error", "text_not_canonical", "document.text", "text must be NFC with LF line endings")
    if text_sha256(text) != document.text_sha256:
        report.add("error", "text_hash", "document.text_sha256", "hash does not match the text")

    layers: dict[str, Layer] = {}
    for index, layer in enumerate(unit.annotation_set.layers):
        where = f"annotation_set.layers[{index}]"
        if layer.layer_id in layers:
            report.add("error", "duplicate_layer", where, f"layer {layer.layer_id!r} is declared twice")
        layers[layer.layer_id] = layer
        if layer.profile not in PROFILES:
            report.add("error", "unknown_profile", f"{where}.profile", f"unknown profile {layer.profile!r}")
        elif layer.profile in HIERARCHICAL_PROFILES and layer.ontology.id not in ontologies:
            report.add(
                "warning",
                "hierarchy_not_checked",
                where,
                f"no ontology {layer.ontology.id!r} given; terms, roots and modifiers were not checked",
            )

    seen: set[str] = set()
    for index, annotation in enumerate(unit.annotation_set.annotations):
        path = f"annotation_set.annotations[{index}]"
        if annotation.annotation_id in seen:
            report.add("error", "duplicate_annotation", path, f"annotation id {annotation.annotation_id!r} is used twice")
        seen.add(annotation.annotation_id)
        layer = layers.get(annotation.layer_id)
        if layer is None:
            report.add("error", "unknown_layer", f"{path}.layer_id", f"layer {annotation.layer_id!r} is not declared")
            continue
        for m, mention in enumerate(annotation.mentions):
            _check_segments(mention.segments, text, f"{path}.mentions[{m}].segments", report)
            _check_segments(mention.cue_segments, text, f"{path}.mentions[{m}].cue_segments", report)
        for m, modifier in enumerate(annotation.modifiers):
            _check_segments(modifier.segments, text, f"{path}.modifiers[{m}].segments", report)
        checker = PROFILES.get(layer.profile)
        if checker is not None:
            report.issues.extend(checker(layer, annotation, ontologies.get(layer.ontology.id), path))


def validate(
    data: Mapping[str, Any] | AnnotatedDocument,
    ontologies: Mapping[str, OntologyView] | None = None,
) -> ValidationReport:
    """Every problem of an annotated document. ``ontologies`` maps ontology ids ("hp")."""
    report = ValidationReport()
    raw = data.model_dump(mode="json") if isinstance(data, AnnotatedDocument) else data
    unit = _parse(raw, report)
    if unit is not None:
        _check_content(unit, ontologies or {}, report)
    return report


def load_annotated_document(
    data: bytes | str | Mapping[str, Any],
    ontologies: Mapping[str, OntologyView] | None = None,
) -> AnnotatedDocument:
    """Parse and validate; raises ``FormatError`` on any error."""
    raw = json.loads(data) if isinstance(data, bytes | str) else data
    report = ValidationReport()
    unit = _parse(raw, report)
    if unit is not None:
        _check_content(unit, ontologies or {}, report)
    if unit is None or not report.ok:
        raise FormatError(report)
    return unit


def dump_json(model: BaseModel) -> bytes:
    """Stable UTF-8 JSON: sorted keys, two-space indent, trailing newline."""
    data = model.model_dump(mode="json")
    return (json.dumps(data, ensure_ascii=False, indent=2, sort_keys=True) + "\n").encode("utf-8")
```

```python
def check_annotation(

    def error(code: str, where: str, message: str) -> None:
        issues.append(Issue("error", code, where, message))

    term = annotation.term_id
    if not _HPO_ID.match(term):
        error("not_an_hpo_id", f"{path}.term_id", f"{term!r} is not an HPO id")

        error("missing_evidence", f"{path}.mentions", "without mentions an annotation needs document_level: true")
    if annotation.mentions and annotation.document_level:
        error("document_level_with_mentions", f"{path}.document_level", "a document-level annotation has no mentions")
    for index, modifier in enumerate(annotation.modifiers):
        if not _HPO_ID.match(modifier.term_id):
            error("not_an_hpo_id", f"{path}.modifiers[{index}].term_id", f"{modifier.term_id!r} is not an HPO id")

    if ontology is None:
        return issues
    if not ontology.is_active(term):
        error("unknown_term", f"{path}.term_id", f"{term} is not an active HPO term")
    elif is_under(ontology, term, CLINICAL_MODIFIER) or is_under(ontology, term, FREQUENCY):
```

```python
"""Bundles: a directory or ZIP with manifest.json and documents/*.json (T1 F3)."""

from __future__ import annotations

import hashlib
import json
import zipfile
from collections.abc import Mapping, Sequence
from datetime import datetime
from pathlib import Path, PurePosixPath

from pydantic import ValidationError

from ontospan._version import SCHEMA_VERSION
from ontospan.models import AnnotatedDocument, Contributor, FileEntry, Layer, Manifest
from ontospan.ontology import OntologyView
from ontospan.report import FormatError, ValidationReport
from ontospan.validate import check_schema, check_version, dump_json, load_annotated_document

MANIFEST = "manifest.json"
MANIFEST_SCHEMA = "bundle-manifest-1.0.json"
MAX_TOTAL_BYTES = 512 * 1024 * 1024


def _layers(units: Sequence[AnnotatedDocument]) -> tuple[Layer, ...]:
    layers: dict[str, Layer] = {}
    for unit in units:
        for layer in unit.annotation_set.layers:
            known = layers.setdefault(layer.layer_id, layer)
            if known != layer:
                raise ValueError(f"layer {layer.layer_id!r} is defined differently in two units")
    return tuple(layers.values())


def write_bundle(
    target: Path,
    units: Sequence[AnnotatedDocument],
    *,
    bundle_id: str,
    created_at: datetime,
    release_name: str | None = None,
    contributors: Sequence[Contributor] = (),
) -> Manifest:
    """Write a bundle; a target ending in .zip becomes a ZIP, otherwise a new directory."""
    if not units:
        raise ValueError("a bundle needs at least one annotated document")
    files = {f"documents/{index:05d}.json": dump_json(unit) for index, unit in enumerate(units)}
    manifest = Manifest(
        schema_version=SCHEMA_VERSION,
        artifact_type="bundle",
        bundle_id=bundle_id,
        release_name=release_name,
        created_at=created_at,
        layers=_layers(units),
        contributors=tuple(contributors),
        files=tuple(FileEntry(path=path, sha256=hashlib.sha256(data).hexdigest()) for path, data in files.items()),
    )
    payload = {MANIFEST: dump_json(manifest), **files}
    if target.suffix == ".zip":
        with zipfile.ZipFile(target, "x", compression=zipfile.ZIP_DEFLATED) as archive:
            for path, data in payload.items():
                archive.writestr(path, data)
    else:
        target.mkdir(parents=True, exist_ok=False)
        for path, data in payload.items():
            (target / path).parent.mkdir(parents=True, exist_ok=True)
            (target / path).write_bytes(data)
    return manifest


def _read_all(source: Path, report: ValidationReport) -> dict[str, bytes]:
    if source.is_dir():
        names = [p.relative_to(source).as_posix() for p in source.rglob("*") if p.is_file()]
        return {name: (source / name).read_bytes() for name in names}
    with zipfile.ZipFile(source) as archive:
        infos = [info for info in archive.infolist() if not info.is_dir()]
        if sum(info.file_size for info in infos) > MAX_TOTAL_BYTES:
            report.add("error", "too_large", str(source), "bundle exceeds 512 MiB uncompressed")
            return {}
        return {info.filename: archive.read(info) for info in infos}


def read_bundle(
    source: Path, ontologies: Mapping[str, OntologyView] | None = None
) -> tuple[Manifest, list[AnnotatedDocument]]:
    """Read and verify a bundle; raises ``FormatError`` with every problem found."""
    report = ValidationReport()
    contents = _read_all(source, report)
    for name in contents:
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts or "\\" in name:
            report.add("error", "unsafe_path", name, "path escapes the bundle")
    if not report.ok:
        raise FormatError(report)
    if MANIFEST not in contents:
        report.add("error", "missing_manifest", MANIFEST, "bundle has no manifest.json")
        raise FormatError(report)

    data = json.loads(contents[MANIFEST])
    if check_version(data.get("schema_version"), report):
        check_schema(data, MANIFEST_SCHEMA, report)
    if not report.ok:
        raise FormatError(report)
    try:
        manifest = Manifest.model_validate(data)
    except ValidationError as exc:
        report.add("error", "model", MANIFEST, str(exc))
        raise FormatError(report) from exc

    listed = {entry.path: entry.sha256 for entry in manifest.files}
    for name in sorted(set(contents) - set(listed) - {MANIFEST}):
        report.add("error", "unlisted_file", name, "file is not listed in the manifest")
    units: list[AnnotatedDocument] = []
    for path, sha in listed.items():
        if path not in contents:
            report.add("error", "missing_file", path, "listed file is missing")
            continue
        if hashlib.sha256(contents[path]).hexdigest() != sha:
            report.add("error", "file_hash", path, "file does not match its hash in the manifest")
            continue
        try:
            units.append(load_annotated_document(contents[path], ontologies))
        except FormatError as exc:
            for issue in exc.report.errors:
                report.add("error", issue.code, f"{path}:{issue.path}", issue.message)
    if report.ok and _layers(units) != manifest.layers:
        report.add("error", "manifest_layers", MANIFEST, "manifest layers differ from the documents' layers")
    if not report.ok:
        raise FormatError(report)
    return manifest, units
```

```python
"""Comparing two annotation sets of the same canonical text (T1 F11).

This is the single implementation that Ontocurator (merge, live agreement)
and the benchmark share.
"""

from __future__ import annotations

from collections import Counter
from collections.abc import Sequence
from dataclasses import dataclass
from typing import Literal

from ontospan.models import Annotation, Mention
from ontospan.ontology import OntologyView
from ontospan.segments import normalize_segment

MatchKind = Literal["equal", "compatible", "axis_conflict", "term_conflict", "a_only", "b_only"]
_RANK: dict[MatchKind, int] = {"equal": 0, "compatible": 1, "axis_conflict": 2, "term_conflict": 3}
Interval = tuple[int, int]


@dataclass(frozen=True)
class MatchGroup:
    kind: MatchKind
    a: Annotation | None
    b: Annotation | None
    ancestor: bool = False
    modifiers_differ: bool = False


@dataclass(frozen=True)
class Agreement:
    strict_f1: float
    tolerant_f1: float
    term_f1: float
    modifier_agreement: float | None
    equal: int
    compatible: int
    axis_conflicts: int
    term_conflicts: int
    a_only: int
    b_only: int


def mention_key(mention: Mention, text: str) -> frozenset[Interval]:
    """A mention as its set of normalised segments."""
    return frozenset(
        interval for seg in mention.segments if (interval := normalize_segment(text, seg)) is not None
    )


def _overlap(x: frozenset[Interval], y: frozenset[Interval]) -> bool:
    return any(s1 < e2 and s2 < e1 for s1, e1 in x for s2, e2 in y)


def _relation(a: Annotation, b: Annotation, text: str) -> Literal["identical", "overlapping", "disjoint"]:
    if not a.mentions and not b.mentions:
        return "identical"
    keys_a = {mention_key(m, text) for m in a.mentions}
    keys_b = {mention_key(m, text) for m in b.mentions}
    if keys_a == keys_b:
        return "identical"
    if any(_overlap(x, y) for x in keys_a for y in keys_b):
        return "overlapping"
    return "disjoint"


def _axes(annotation: Annotation) -> tuple[tuple[str, str], ...]:
    return tuple(sorted((axis, value.value) for axis, value in annotation.axes.items()))


def _classify(a: Annotation, b: Annotation, text: str) -> MatchKind | None:
    relation = _relation(a, b, text)
    same_term = a.term_id == b.term_id
    if relation == "disjoint" or (not a.mentions and not b.mentions and not same_term):
        return None
    if same_term and _axes(a) == _axes(b):
        return "equal" if relation == "identical" else "compatible"
    return "axis_conflict" if same_term else "term_conflict"


def _position(annotation: Annotation, text: str) -> tuple[int, str]:
    starts = [interval[0] for m in annotation.mentions for interval in mention_key(m, text)]
    return (min(starts) if starts else -1, annotation.annotation_id)


def match_annotations(
    a: Sequence[Annotation],
    b: Sequence[Annotation],
    text: str,
    ontology: OntologyView | None = None,
) -> list[MatchGroup]:
    """Pair the annotations of two sets, best relation first; the rest stays unpaired."""
    left = sorted(a, key=lambda x: _position(x, text))
    right = sorted(b, key=lambda x: _position(x, text))
    candidates: list[tuple[int, int, int, MatchKind]] = []
    for i, x in enumerate(left):
        for j, y in enumerate(right):
            kind = _classify(x, y, text)
            if kind is not None:
                candidates.append((_RANK[kind], i, j, kind))
    candidates.sort()
    used_a: set[int] = set()
    used_b: set[int] = set()
    groups: list[MatchGroup] = []
    for _, i, j, kind in candidates:
        if i in used_a or j in used_b:
            continue
        used_a.add(i)
        used_b.add(j)
        x, y = left[i], right[j]
        ancestor = (
            kind == "term_conflict"
            and ontology is not None
            and (x.term_id in ontology.ancestors(y.term_id) or y.term_id in ontology.ancestors(x.term_id))
        )
        modifiers_differ = x.term_id == y.term_id and (
            {m.term_id for m in x.modifiers} != {m.term_id for m in y.modifiers}
        )
        groups.append(MatchGroup(kind, x, y, ancestor, modifiers_differ))
    groups.extend(MatchGroup("a_only", x, None) for i, x in enumerate(left) if i not in used_a)
    groups.extend(MatchGroup("b_only", None, y) for j, y in enumerate(right) if j not in used_b)
    return groups


def _f1(true_positives: int, n_a: int, n_b: int) -> float:
    return 1.0 if n_a + n_b == 0 else 2 * true_positives / (n_a + n_b)


def agreement(
    a: Sequence[Annotation],
    b: Sequence[Annotation],
    text: str,
    ontology: OntologyView | None = None,
) -> Agreement:
    """Pairwise agreement of two annotation sets: always strict and tolerant (T1 F11)."""
    groups = match_annotations(a, b, text, ontology)
    count = Counter(group.kind for group in groups)
    terms_a = {(x.term_id, _axes(x)) for x in a}
    terms_b = {(y.term_id, _axes(y)) for y in b}
    paired = [g for g in groups if g.kind in ("equal", "compatible")]
    return Agreement(
        strict_f1=_f1(count["equal"], len(a), len(b)),
        tolerant_f1=_f1(count["equal"] + count["compatible"], len(a), len(b)),
        term_f1=_f1(len(terms_a & terms_b), len(terms_a), len(terms_b)),
        modifier_agreement=(sum(not g.modifiers_differ for g in paired) / len(paired)) if paired else None,
        equal=count["equal"],
        compatible=count["compatible"],
        axis_conflicts=count["axis_conflict"],
        term_conflicts=count["term_conflict"],
        a_only=count["a_only"],
        b_only=count["b_only"],
    )
```

Check of the agreement test numbers: a1↔b1 is equal ("Hypotonie" vs "Hypotonie,"), a2↔b2 is compatible, and a3 is a-only. Strict F1 = 2·1/(3+2) and tolerant F1 = 2·2/5. Term F1: the sets {(HYPOTONIA), (SEIZURE), (FEVER)} and {(HYPOTONIA), (SEIZURE)} share 2 elements, which gives 2·2/5. Modifier agreement is 1.0.

```python
"""Phenopacket v2 export (T1 F9): derived, lossy, with a report of what was left out.

The packet is built as protobuf-JSON (camelCase) without depending on the
``phenopackets`` library; ``validate_phenopacket`` uses it when installed
(extra ``[phenopacket]``).
"""

from __future__ import annotations

from dataclasses import dataclass
from datetime import UTC, datetime
from typing import Any

from ontospan._version import __version__
from ontospan.models import AnnotatedDocument, Annotation
from ontospan.ontology import OntologyView, is_under
from ontospan.profiles.hpo import FREQUENCY, NAME as HPO_PROFILE, ONSET, SEVERITY

MANUAL = {"id": "ECO:0000218", "label": "manual assertion"}
AUTOMATIC = {"id": "ECO:0007636", "label": "computational evidence used in automatic assertion"}


@dataclass
class ExportReport:
    exported: int = 0
    omitted_uncertain: int = 0
    omitted_not_patient: int = 0
    uncertain_included: int = 0
    frequency_modifiers: int = 0


def _term(ontology: OntologyView, term_id: str) -> dict[str, str]:
    return {"id": term_id, "label": ontology.label(term_id) or term_id}


def _value(annotation: Annotation, axis: str) -> str | None:
    given = annotation.axes.get(axis)
    return given.value if given is not None else None


def _feature(
    annotation: Annotation,
    ontology: OntologyView,
    reference: str,
    report: ExportReport,
) -> dict[str, Any]:
    feature: dict[str, Any] = {"type": _term(ontology, annotation.term_id)}
    if _value(annotation, "assertion") == "absent":
        feature["excluded"] = True
    others: list[dict[str, str]] = []
    for modifier in annotation.modifiers:
        term = modifier.term_id
        if "severity" not in feature and is_under(ontology, term, SEVERITY):
            feature["severity"] = _term(ontology, term)
        elif "onset" not in feature and is_under(ontology, term, ONSET):
            feature["onset"] = {"ontologyClass": _term(ontology, term)}
        else:
            if is_under(ontology, term, FREQUENCY):
                report.frequency_modifiers += 1
            others.append(_term(ontology, term))
    if others:
        feature["modifiers"] = others
    code = AUTOMATIC if annotation.provenance.actor.kind == "tool" else MANUAL
    evidence: dict[str, Any] = {"evidenceCode": code, "reference": {"id": reference}}
    if _value(annotation, "assertion") == "uncertain":
        evidence["reference"]["description"] = "assertion: uncertain (suspected, not confirmed)"
    feature["evidence"] = [evidence]
    return feature


def to_phenopacket(
    unit: AnnotatedDocument,
    ontology: OntologyView,
    *,
    include_uncertain: bool = False,
    bundle_id: str | None = None,
    created_at: datetime | None = None,
    phenopacket_id: str | None = None,
) -> tuple[dict[str, Any], ExportReport]:
    """One Phenopacket for one annotated document, from its hpo-phenotype layers."""
    report = ExportReport()
    document = unit.document
    layers = {layer.layer_id: layer for layer in unit.annotation_set.layers if layer.profile == HPO_PROFILE}
    features: list[dict[str, Any]] = []
    for annotation in unit.annotation_set.annotations:
        if annotation.layer_id not in layers:
            continue
        if _value(annotation, "experiencer") != "patient":
            report.omitted_not_patient += 1
            continue
        if _value(annotation, "assertion") == "uncertain":
            if not include_uncertain:
                report.omitted_uncertain += 1
                continue
            report.uncertain_included += 1
        reference = f"urn:ontospan:{bundle_id or '-'}:{document.document_id}:{annotation.annotation_id}"
        features.append(_feature(annotation, ontology, reference, report))
        report.exported += 1

    created = (created_at or datetime.now(UTC)).astimezone(UTC)
    resources = [
        {
            "id": "hp",
            "name": "human phenotype ontology",
            "url": "http://purl.obolibrary.org/obo/hp.owl",
            "version": layer.ontology.version,
            "namespacePrefix": "HP",
            "iriPrefix": "http://purl.obolibrary.org/obo/HP_",
        }
        for layer in list(layers.values())[:1]
    ]
    packet: dict[str, Any] = {"id": phenopacket_id or document.document_id}
    if document.subject_id:
        packet["subject"] = {"id": document.subject_id}
    packet["phenotypicFeatures"] = features
    packet["metaData"] = {
        "created": created.strftime("%Y-%m-%dT%H:%M:%SZ"),
        "createdBy": f"ontospan {__version__}",
        "resources": resources,
        "phenopacketSchemaVersion": "2.0",
    }
    return packet, report


def validate_phenopacket(packet: dict[str, Any]) -> None:
    """Parse into the official classes; raises on unknown or malformed fields."""
    from google.protobuf.json_format import ParseDict
    from phenopackets import Phenopacket

    ParseDict(packet, Phenopacket())
```
